--- NEW FILE apps/web/lib/billing/recovery-flag.ts ---
import "server-only";
/** Server-side master switch for scheduled billing recovery WRITES. Default OFF. Strict: only the
 *  literal string "true" enables; unset/empty/"1"/"TRUE"/anything else = OFF (fail closed).
 *  Independent of CRON_SECRET (auth) and of BILLING_RECOVERY_SCHEDULE_ENABLED (GitHub cadence). */
export function isBillingRecoveryEnabled(env: NodeJS.ProcessEnv = process.env): boolean {
  return (env.BILLING_RECOVERY_ENABLED ?? "").trim() === "true";
}
